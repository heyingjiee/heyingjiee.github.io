节流、防抖

```JS
// 节流。第一次触发后，延迟时间内再次触发无效
function throttle(fn,delay){
  let valid=true
  return functin(){
    if(!valid){
      return 
    }
    valid=false
    setTimeout(()=>{
      fn()
      valid=true
    },delay)
  }
}

// 防抖。第一次触发后，延迟时间内再次触发重新计时
function debounce(fn,delay){
  let time=null
  return function(){
    if(time){
      clearTimeout(time)
    }
    setTimeout(fn,delay)
  }
}
```
